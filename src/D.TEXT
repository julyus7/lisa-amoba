unit MSDraw;INTERFACEuses (*$U UnitStd.obj *)        UnitStd,     (*$U UnitHz.obj *)         UnitHz,     (*$U Storage.obj *)        Storage,     (*$U QuickDraw.obj *)      QuickDraw,     (*$U Fontmgr.obj *)        FontMgr,     (*$U dbenv.obj *)          dbenv,     (*$U FEdec.obj *)          FEdec,     (*$U fld.obj *)            fieldedit,     (*$U SysCall.obj *)        SysCall,     (*$U UnitFile.obj *)       UnitFile,     (*$U PMDecl.obj *)         PMDecl,     (*$U PrStdInfo.obj *)      PrStdInfo,     (*$U PrPublic.obj *)       PrPublic,     (*$U wm.events.obj *)      Events,     (*$U wm.folders.obj *)     Folders,     (*$U wm.menus.obj *)       Menus,     (*$U AlertMgr.obj *)       AlertMgr,     (*$U WMLstd.obj *)         WMLstd,     (*$U WMlCrs.obj *)         WMLCrs,     (*$U UnitFmt.obj *)        UnitFmt,     (*$U UnitCs.obj *)         UnitCs,     (*$U UnitFf.obj *)         UnitFf,     (*$U scrap.obj *)          Scrap,     (*$U FilerComm.obj *)      FilerComm,     (*$U G.OBJ *) MSGlobals;procedure ComputeLayout;procedure CellRect (c, r : integer; var box : Rect);function PointToCell (pt : Point; var c, r : integer) : boolean;procedure FaceRect (var box : Rect);procedure DrawCell (c, r : integer);procedure DrawDirtyCells;procedure DrawBoard;procedure DrawCounter;procedure DrawTimer;procedure DrawFace (pressed : boolean);procedure DrawStatusBar;procedure DrawContents (window : WindowPtr);procedure RenderImage (port : GrafPtr);procedure FrameImage (port : GrafPtr);procedure DrawWindowScrap (normal : boolean);IMPLEMENTATIONconst   wellH    = 26;   wellPad  = 5;   faceW    = 33;   faceH    = 22;   qmW      = 11;   qmH      = 11;function TargetPort : GrafPtr;begin   if imagePort <> nil then TargetPort := imagePort   else TargetPort := myWindowend;procedure ComputeLayout;begin   contentW := 2 * winMargin + myDoc^.cols * cellW;   contentH := boardTop + myDoc^.rows * cellH + winMarginend;procedure CellRect ;begin   SetRect(box, boardLeft + (c - 1) * cellW,                boardTop  + (r - 1) * cellH,                boardLeft + c * cellW,                boardTop  + r * cellH)end;function PointToCell ;begin   PointToCell := false;   if (pt.h >= boardLeft) and (pt.v >= boardTop) then begin      c := (pt.h - boardLeft) div cellW + 1;      r := (pt.v - boardTop)  div cellH + 1;      if (c >= 1) and (c <= myDoc^.cols) and (r >= 1) and (r <= myDoc^.rows) then         PointToCell := true   endend;procedure FaceRect ;var cx, cy : integer;begin   cx := contentW div 2;   cy := winMargin + statusH div 2;   SetRect(box, cx - faceW div 2, cy - faceH div 2,                cx - faceW div 2 + faceW, cy - faceH div 2 + faceH)end;procedure DrawReadout (n : integer; var well : Rect);var s     : Str255;    m     : integer;    fi    : FontInfo;    x, y  : integer;    inner : Rect;begin   if n > 999 then n := 999;   if n < -99 then n := -99;   s := '000';   if n < 0 then begin      m := -n;      s[1] := '-';      s[2] := chr(ord('0') + m div 10);      s[3] := chr(ord('0') + m mod 10)   end   else begin      s[1] := chr(ord('0') + n div 100);      s[2] := chr(ord('0') + (n div 10) mod 10);      s[3] := chr(ord('0') + n mod 10)   end;   TextFont(fntStatus);   TextFace([]);   GetFontInfo(fi);   inner := well;   InsetRect(inner, 2, 2);   EraseRect(inner);   x := well.right - wellPad - StringWidth(s);   y := well.top + (well.bottom - well.top + fi.ascent - fi.descent) div 2;   MoveTo(x, y);   TextMode(srcOr);   DrawString(s)end;procedure DrawWell (var well : Rect);var inner : Rect;begin   EraseRect(well);   PenNormal;   FrameRect(well);   inner := well;   InsetRect(inner, 1, 1);   MoveTo(inner.left, inner.bottom - 1);   LineTo(inner.left, inner.top);   LineTo(inner.right - 1, inner.top)end;procedure CounterWell (var well : Rect);var top : integer;begin   top := winMargin + (statusH - wellH) div 2;   TextFont(fntStatus);   TextFace([]);   SetRect(well, winMargin + 4, top,                 winMargin + 4 + StringWidth('000') + 2 * wellPad + 2, top + wellH)end;procedure TimerWell (var well : Rect);var top, w : integer;begin   top := winMargin + (statusH - wellH) div 2;   TextFont(fntStatus);   TextFace([]);   w := StringWidth('000') + 2 * wellPad + 2;   SetRect(well, contentW - winMargin - 4 - w, top,                 contentW - winMargin - 4, top + wellH)end;procedure DrawMineGlyph (var box : Rect);var cx, cy : integer;    body   : Rect;    glint  : Rect;begin   cx := (box.left + box.right) div 2;   cy := (box.top + box.bottom) div 2;   PenNormal;   MoveTo(cx - 9, cy);      LineTo(cx + 9, cy);   MoveTo(cx, cy - 5);      LineTo(cx, cy + 5);   MoveTo(cx - 7, cy - 5);  LineTo(cx + 7, cy + 5);   MoveTo(cx - 7, cy + 5);  LineTo(cx + 7, cy - 5);   SetRect(body, cx - 6, cy - 4, cx + 6, cy + 4);   PaintOval(body);   SetRect(glint, cx - 3, cy - 2, cx - 1, cy - 1);   FillRect(glint, white)end;procedure DrawFlagGlyph (var box : Rect; whiteInk : boolean);var staffX : integer;    i      : integer;begin   PenNormal;   if whiteInk then PenPat(white);   staffX := box.left + cellW div 2 + 3;   MoveTo(staffX, box.top + 3);   LineTo(staffX, box.bottom - 4);   MoveTo(staffX - 4, box.bottom - 4);   LineTo(staffX + 3, box.bottom - 4);   for i := 0 to 2 do begin      MoveTo(staffX - 8 + 2 * i, box.top + 3 + i);      LineTo(staffX - 1, box.top + 3 + i)   end;   PenNormalend;procedure DrawQueryGlyph (var box : Rect);var gx, gy : integer;    r      : Rect;begin   PenNormal;   gx := box.left + (cellW - qmW) div 2;   gy := box.top + (cellH - qmH) div 2;   MoveTo(gx + 3, gy);      LineTo(gx + 7, gy);   MoveTo(gx + 1, gy + 1);  LineTo(gx + 9, gy + 1);   SetRect(r, gx, gy + 2, gx + 3, gy + 4);          PaintRect(r);   SetRect(r, gx + 8, gy + 2, gx + 11, gy + 5);     PaintRect(r);   MoveTo(gx + 5, gy + 5);  LineTo(gx + 9, gy + 5);   MoveTo(gx + 4, gy + 6);  LineTo(gx + 6, gy + 6);   SetRect(r, gx + 4, gy + 9, gx + 7, gy + 11);   PaintRect(r)end;procedure DrawDigitGlyph (n : integer; var box : Rect);var ch : char;    fi : FontInfo;begin   ch := chr(ord('0') + n);   TextFont(fntCell);   if n >= 4 then TextFace([bold])             else TextFace([]);   GetFontInfo(fi);   MoveTo(box.left + (cellW - CharWidth(ch)) div 2,          box.top + (cellH + fi.ascent - fi.descent) div 2);   TextMode(srcOr);   DrawChar(ch);   TextFace([])end;procedure DrawCell ;var box   : Rect;    inner : Rect;    showMine : boolean;begin   if TargetPort = nil then exit(DrawCell);   SetPort(TargetPort);   CellRect(c, r, box);   with myDoc^ do begin      if cells[r, c].isRevealed and not cells[r, c].isMine then begin         EraseRect(box);         PenNormal;         PenPat(gray);         MoveTo(box.left, box.bottom - 1);         LineTo(box.right - 1, box.bottom - 1);         LineTo(box.right - 1, box.top);         PenPat(black);         if cells[r, c].adjMines > 0 then            DrawDigitGlyph(cells[r, c].adjMines, box)      end      else if cells[r, c].isRevealed and cells[r, c].isMine then begin         EraseRect(box);         DrawMineGlyph(box);         inner := box;         InsetRect(inner, 1, 1);         InvertRect(inner)      end      else begin         FillRect(box, ltGray);         PenNormal;         PenPat(white);         MoveTo(box.left, box.bottom - 2);         LineTo(box.left, box.top);         LineTo(box.right - 2, box.top);         PenPat(black);         MoveTo(box.left, box.bottom - 1);         LineTo(box.right - 1, box.bottom - 1);         LineTo(box.right - 1, box.top);         showMine := false;         if state = msLost then            if cells[r, c].isMine and (cells[r, c].playerMark <> pmFlag) then               showMine := true;         if showMine then            DrawMineGlyph(box)         else if cells[r, c].playerMark = pmFlag then begin            if (state = msLost) and cells[r, c].isMine then begin               FillRect(box, white);               inner := box;               InsetRect(inner, 1, 1);               FillRect(inner, black);               DrawFlagGlyph(box, true)            end            else               DrawFlagGlyph(box, false)         end         else if cells[r, c].playerMark = pmQuery then            DrawQueryGlyph(box)      end   endend;procedure DrawDirtyCells;var c, r : integer;begin   if myWindow = nil then begin      ClearDirty;      exit(DrawDirtyCells)   end;   SetPort(myWindow);   for r := 1 to myDoc^.rows do      for c := 1 to myDoc^.cols do         if dirtyCell[r, c] then begin            dirtyCell[r, c] := false;            DrawCell(c, r)         endend;procedure DrawBoard;var c, r  : integer;    frame : Rect;begin   if TargetPort = nil then exit(DrawBoard);   SetPort(TargetPort);   for r := 1 to myDoc^.rows do      for c := 1 to myDoc^.cols do         DrawCell(c, r);   if not picRecording then      ClearDirty;   SetRect(frame, boardLeft - 1, boardTop - 1,                  boardLeft + myDoc^.cols * cellW + 1,                  boardTop + myDoc^.rows * cellH + 1);   PenNormal;   FrameRect(frame)end;procedure DrawCounter;var well : Rect;begin   if TargetPort = nil then exit(DrawCounter);   SetPort(TargetPort);   CounterWell(well);   DrawWell(well);   DrawReadout(myDoc^.mineCount - myDoc^.flagsPlaced, well)end;procedure DrawTimer;var well : Rect;begin   if TargetPort = nil then exit(DrawTimer);   SetPort(TargetPort);   TimerWell(well);   DrawReadout(myDoc^.elapsed, well)end;procedure DrawFace ;var btn, face, ii : Rect;    cx, cy : integer;begin   if TargetPort = nil then exit(DrawFace);   SetPort(TargetPort);   FaceRect(btn);   EraseRoundRect(btn, 9, 6);   PenNormal;   FrameRoundRect(btn, 9, 6);   cx := (btn.left + btn.right) div 2;   cy := (btn.top + btn.bottom) div 2;   SetRect(face, cx - 9, cy - 6, cx + 9, cy + 6);   FrameOval(face);   if pressed then begin      SetRect(ii, cx - 5, cy - 3, cx - 3, cy - 1);  PaintOval(ii);      SetRect(ii, cx + 3, cy - 3, cx + 5, cy - 1);  PaintOval(ii);      SetRect(ii, cx - 2, cy + 1, cx + 2, cy + 4);  FrameOval(ii)   end   else if myDoc^.state = msLost then begin      PenNormal;      MoveTo(cx - 5, cy - 3);  LineTo(cx - 2, cy - 1);      MoveTo(cx - 2, cy - 3);  LineTo(cx - 5, cy - 1);      MoveTo(cx + 2, cy - 3);  LineTo(cx + 5, cy - 1);      MoveTo(cx + 5, cy - 3);  LineTo(cx + 2, cy - 1);      SetRect(ii, cx - 3, cy, cx + 3, cy + 6);      FrameArc(ii, 270, 180)   end   else if myDoc^.state = msWon then begin      SetRect(ii, cx - 7, cy - 4, cx - 1, cy - 1);  PaintRect(ii);      SetRect(ii, cx + 1, cy - 4, cx + 7, cy - 1);  PaintRect(ii);      MoveTo(cx - 1, cy - 3);  LineTo(cx + 1, cy - 3);      SetRect(ii, cx - 4, cy - 1, cx + 4, cy + 5);      FrameArc(ii, 90, 180)   end   else begin      SetRect(ii, cx - 5, cy - 3, cx - 3, cy - 1);  PaintOval(ii);      SetRect(ii, cx + 3, cy - 3, cx + 5, cy - 1);  PaintOval(ii);      SetRect(ii, cx - 4, cy - 2, cx + 4, cy + 4);      FrameArc(ii, 90, 180)   endend;procedure DrawStatusBar;var well : Rect;begin   if TargetPort = nil then exit(DrawStatusBar);   SetPort(TargetPort);   DrawCounter;   TimerWell(well);   DrawWell(well);   DrawReadout(myDoc^.elapsed, well);   DrawFace(false)end;procedure DrawContents ;begin   RenderImage(window)end;procedure RenderImage ;var all : Rect;begin   if port = nil then exit(RenderImage);   imagePort := port;   SetPort(port);   SetRect(all, 0, 0, contentW, contentH);   ClipRect(all);   EraseRect(all);   DrawStatusBar;   DrawBoard;   imagePort := nilend;procedure FrameImage ;var all : Rect;begin   if port = nil then exit(FrameImage);   SetPort(port);   SetRect(all, 0, 0, contentW, contentH);   PenNormal;   FrameRect(all)end;procedure DrawWindowScrap ;var err : integer;begin   if normal then BeginUpdate(scrapFolder);   if ScrapUG in CurrScrapSet then      DrawUScrap(err)   else if ScrapFE in CurrScrapSet then      DrawScrap(err)   else begin      SetPort(scrapFolder);      EraseRect(scrapFolder^.portRect)   end;   if normal then EndUpdate(scrapFolder)end;end.
