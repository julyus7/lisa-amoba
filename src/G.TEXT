boolean);IMPLEMENTATIONconst   clkWaitMs = 100;   clkTries  = 16;var   clkBaseSecs : longint;   clkRuns     : boolean;procedure MarkDirty ;begin   if (c >= 1) and (c <= maxCols) and (r >= 1) and (r <= maxRows) then      dirtyCell[r, c] := trueend;procedure ClearDirty;var c, r : integer;begin   for r := 1 to maxRows do      for c := 1 to maxCols do         dirtyCell[r, c] := falseend;function ClockSecs ;var err : integer;    t   : Time_Rec;begin   Get_Time(err, t);   ClockSecs := ((ord4(t.day) * 24 + t.hour) * 60 + t.minute) * 60 + t.secondend;procedure StartClockCheck;begin   clkRuns     := false;   clkBaseSecs := ClockSecsend;function ClockIsRunning ;var tries : integer;    err   : integer;    iv    : timestmp_interval;    nowT  : Time_Rec;begin   if clkRuns then begin      ClockIsRunning := true;      exit(ClockIsRunning)   end;   if ClockSecs <> clkBaseSecs then begin      clkRuns := true;      ClockIsRunning := true;      exit(ClockIsRunning)   end;   tries   := 0;   iv.sec  := 0;   iv.msec := clkWaitMs;   Get_Time(err, nowT);   repeat      Delay_Time(err, iv, nowT);      tries   := tries + 1;      clkRuns := ClockSecs <> clkBaseSecs   until clkRuns or (tries >= clkTries);   clkBaseSecs := ClockSecs;   ClockIsRunning := clkRunsend;procedure PutUpScrapMenus;begin   ClearMenuBar; theEvent, fx);   case fx.theFlrOp of      fcNone :         wantsOpen := true;      fcResume :         TellFiler(err, docClosd, internalError, theEvent.who);      fcClose, fcSuspend, fcShred :         begin            myWindow := nil;            TellFiler(err, docClosd, allOK, theEvent.who)         end;      fcCopy, fcPut :         TellFiler(err, docNotXfered, internalError, theEvent.who);      fcDfClose :         TellFiler(err, dfClosed, allOK, theEvent.who);      fcTerminate :         toldToDie := true;      otherwise   endend;end.
