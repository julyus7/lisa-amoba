unit MSDialog;INTERFACEuses (*$U UnitStd.obj *)        UnitStd,     (*$U UnitHz.obj *)         UnitHz,     (*$U Storage.obj *)        Storage,     (*$U QuickDraw.obj *)      QuickDraw,     (*$U Fontmgr.obj *)        FontMgr,     (*$U dbenv.obj *)          dbenv,     (*$U FEdec.obj *)          FEdec,     (*$U fld.obj *)            fieldedit,     (*$U SysCall.obj *)        SysCall,     (*$U UnitFile.obj *)       UnitFile,     (*$U PMDecl.obj *)         PMDecl,     (*$U PrStdInfo.obj *)      PrStdInfo,     (*$U PrPublic.obj *)       PrPublic,     (*$U wm.events.obj *)      Events,     (*$U wm.folders.obj *)     Folders,     (*$U wm.menus.obj *)       Menus,     (*$U AlertMgr.obj *)       AlertMgr,     (*$U WMLstd.obj *)         WMLstd,     (*$U WMlCrs.obj *)         WMLCrs,     (*$U UnitFmt.obj *)        UnitFmt,     (*$U UnitCs.obj *)         UnitCs,     (*$U UnitFf.obj *)         UnitFf,     (*$U scrap.obj *)          Scrap,     (*$U FilerComm.obj *)      FilerComm,     (*$U G.OBJ *) MSGlobals,     (*$U D.OBJ *)    MSDraw;procedure InitDialogs (var err : integer);function DoCustomDialog (var w, h, m : integer) : boolean;function DoNameDialog (var winnerName : Str255) : boolean;IMPLEMENTATIONconst   numFldLen  = 3;   nameFldLen = 20;   cusPanW  = 190;   cusPanH  = 150;   namPanW  = 190;   namPanH  = 104;   rowH    = 26;   fBoxH   = 18;   numBoxW = 46;   namBoxW = 158;   btnW    = 62;   btnH    = 18;var   fldW, fldH, fldM : hndField;   fldName          : hndField;   dlgFSH           : hndFState;   curFld    : hndField;   blinkT    : integer;   nextBlink : longint;   panelR    : Rect;   isCustom  : boolean;   okR, cancelR : Rect;   lastFldTime : longint;   lastFldHnd  : hndField;procedure GetPhrase (n : integer; var s : Str255);begin   s := '';   GetAlert(allAlerts, n, @s)end;procedure FieldText (hf : hndField; var s : Str255);var i, len, b : integer;begin   len := hf^^.curLen;   if len > 255 then len := 255;   s := '';   s[0] := chr(len);   for i := 1 to len do begin      b := hf^^.curValue^^[i - 1];      if b < 0 then b := b + 256;      s[i] := chr(b)   endend;function TextToInt (var s : Str255; var n : integer) : boolean;var i : integer;begin   TextToInt := false;   n := 0;   if (length(s) < 1) or (length(s) > numFldLen) then exit(TextToInt);   for i := 1 to length(s) do begin      if (s[i] < '0') or (s[i] > '9') then exit(TextToInt);      n := n * 10 + (ord(s[i]) - ord('0'))   end;   TextToInt := trueend;procedure IntToText (n : integer; var s : Str255);begin   s := '';   if n >= 100 then s := concat(s, ' ');   if n >= 10  then s := concat(s, ' ');   s := concat(s, ' ');   if n >= 100 then s[length(s) - 2] := chr(ord('0') + n div 100);   if n >= 10  then s[length(s) - 1] := chr(ord('0') + (n div 10) mod 10);   s[length(s)] := chr(ord('0') + n mod 10)end;procedure StuffField (hf : hndField; var s : Str255);var i, ferr : integer;begin   ClearField(hf, nil, ferr);   hf^^.curLen := length(s);   for i := 1 to length(s) do      hf^^.curValue^^[i - 1] := ord(s[i])end;procedure DropFocus;begin   if curFld <> nil then begin      if myWindow <> nil then begin         SetPort(myWindow);         ClipRect(myWindow^.portRect);         ExitField(curFld, dlgFSH);         DrawField(curFld, nil, false, true)      end;      curFld := nil;      blinkT := -1   endend;procedure SetFocusAll (hf : hndField);begin   DropFocus;   SetPort(myWindow);   ClipRect(myWindow^.portRect);   curFld := hf;   SelectAll(curFld, dlgFSH, blinkT);   if blinkT > 0 then nextBlink := Time + blinkTend;procedure LayoutPanel;var pw, ph : integer;    box    : Rect;   procedure OneRow (rowIx : integer; hf : hndField; boxW : integer);   var fb : Rect;   begin      SetRect(fb, panelR.right - 16 - boxW,                  panelR.top + 30 + rowIx * rowH,                  panelR.right - 16,                  panelR.top + 30 + rowIx * rowH + fBoxH);      box := fb;      InsetRect(box, 3, 2);      SetFCoords(hf, nil, box)   end;begin   if isCustom then begin pw := cusPanW; ph := cusPanH end               else begin pw := namPanW; ph := namPanH end;   SetRect(panelR, (contentW - pw) div 2,                   (contentH - ph) div 2,                   (contentW - pw) div 2 + pw,                   (contentH - ph) div 2 + ph);   if isCustom then begin      OneRow(0, fldW, numBoxW);      OneRow(1, fldH, numBoxW);      OneRow(2, fldM, numBoxW)   end   else      OneRow(0, fldName, namBoxW);   SetRect(cancelR, panelR.left + 14, panelR.bottom - btnH - 10,                    panelR.left + 14 + btnW, panelR.bottom - 10);   SetRect(okR, panelR.right - btnW - 14, panelR.bottom - btnH - 10,                panelR.right - 14, panelR.bottom - 10)end;procedure DrawPanel;var s     : Str255;    fi    : FontInfo;    inner : Rect;    y     : integer;   procedure FieldBoxAround (hf : hndField);   var fb : Rect;   begin      fb := hf^^.coords;      InsetRect(fb, -3, -2);      PenNormal;      FrameRect(fb)   end;   procedure LabelForRow (rowIx : integer; phrase : integer);   var ls : Str255;   begin      GetPhrase(phrase, ls);      MoveTo(panelR.left + 14,             panelR.top + 30 + rowIx * rowH + fBoxH - 5);      DrawString(ls)   end;   procedure OneButton (var r : Rect; capIx : integer; isDefault : boolean);   var cap : Str255;       big : Rect;   begin      cap := '';      GetButn(capIx, @cap);      EraseRoundRect(r, 9, 6);      PenNormal;      FrameRoundRect(r, 9, 6);      if isDefault then begin         big := r;         InsetRect(big, -3, -2);         FrameRoundRect(big, 12, 8)      end;      MoveTo((r.left + r.right - StringWidth(cap)) div 2,             (r.top + r.bottom + fi.ascent - fi.descent) div 2);      DrawString(cap)   end;begin   SetPort(myWindow);   ClipRect(myWindow^.portRect);   TextFont(fntLabel);   TextFace([]);   GetFontInfo(fi);   EraseRect(panelR);   PenNormal;   FrameRect(panelR);   inner := panelR;   InsetRect(inner, 2, 2);   FrameRect(inner);   if isCustom then GetPhrase(phCustomTitle, s)               else GetPhrase(phNameTitle, s);   TextFace([bold]);   y := panelR.top + 8 + fi.ascent;   MoveTo((panelR.left + panelR.right - StringWidth(s)) div 2, y);   DrawString(s);   TextFace([]);   if isCustom then begin      LabelForRow(0, phWidth);      LabelForRow(1, phHeight);      LabelForRow(2, phMines);      FieldBoxAround(fldW);      FieldBoxAround(fldH);      FieldBoxAround(fldM);      if curFld = fldW then DrawField(fldW, dlgFSH, true, true)                       else DrawField(fldW, nil, false, true);      if curFld = fldH then DrawField(fldH, dlgFSH, true, true)                       else DrawField(fldH, nil, false, true);      if curFld = fldM then DrawField(fldM, dlgFSH, true, true)                       else DrawField(fldM, nil, false, true)   end   else begin      FieldBoxAround(fldName);      if curFld = fldName then DrawField(fldName, dlgFSH, true, true)                          else DrawField(fldName, nil, false, true);      GetPhrase(phNamePrompt, s);      MoveTo((panelR.left + panelR.right - StringWidth(s)) div 2,             panelR.top + 30 + fBoxH + 6 + fi.ascent);      DrawString(s)   end;   OneButton(cancelR, 0, false);   OneButton(okR, 1, true)end;procedure SetDlgMenuState (dialogUp : boolean);begin   if dialogUp then begin      DisableMenu(myMenus[mFile]);      DisableMenu(myMenus[mGame]);      DisableMenu(myMenus[mLevel]);      EnableMenu(myMenus[mEdit]);      EnableItem(myMenus[mEdit], miUndo);      EnableItem(myMenus[mEdit], miCut);      EnableItem(myMenus[mEdit], miCopy);      EnableItem(myMenus[mEdit], miPaste)   end   else begin      EnableMenu(myMenus[mFile]);      EnableMenu(myMenus[mGame]);      EnableMenu(myMenus[mLevel]);      DisableItem(myMenus[mEdit], miUndo);      DisableItem(myMenus[mEdit], miCut);      EnableItem(myMenus[mEdit], miCopy);      DisableItem(myMenus[mEdit], miPaste)   end;   DrawMenuBarend;procedure PutUpDlgMenus;var mm : integer;begin   ClearMenuBar;   for mm := mFile to lastBarMenu do      InsertMenu(myMenus[mm], 0);   SetDlgMenuState(true)end;procedure DoDlgEditCommand (item : integer);var ferr : integer;    limit : integer;begin   if curFld = nil then exit(DoDlgEditCommand);   SetPort(myWindow);   ClipRect(myWindow^.portRect);   case item of      miUndo  : UndoEdit(curFld, dlgFSH, ferr);      miCut   : CutCopyField(curFld, dlgFSH, true, true, ferr);      miCopy  : CutCopyField(curFld, dlgFSH, false, true, ferr);      miPaste : begin                   limit := curFld^^.maxLen;                   PasteField(curFld, dlgFSH, ferr, false, limit)                end;      otherwise ferr := 0   end;   if ferr <> 0 then BeepAlert(1);   DrawField(curFld, dlgFSH, true, true)end;var   dlgDone     : boolean;   dlgAccepted : boolean;function TrackDlgButton (var r : Rect) : boolean;var pt : Point;begin   SetPort(myWindow);   InvertRoundRect(r, 9, 6);   while StillDown do      LetOthersRun;   InvertRoundRect(r, 9, 6);   GetMouse(pt);   TrackDlgButton := PtInRect(pt, r)end;function FieldFromPoint (pt : Point) : hndField;   function Hit (hf : hndField) : boolean;   var fb : Rect;   begin      fb := hf^^.coords;      InsetRect(fb, -3, -2);      Hit := PtInRect(pt, fb)   end;begin   FieldFromPoint := nil;   if isCustom then begin      if Hit(fldW) then FieldFromPoint := fldW      else if Hit(fldH) then FieldFromPoint := fldH      else if Hit(fldM) then FieldFromPoint := fldM   end   else      if Hit(fldName) then FieldFromPoint := fldNameend;procedure DlgMouseInField (hf : hndField; pt : Point);var nbr    : Rect;    mpt    : Point;    isDbl  : boolean;begin   SetPort(myWindow);   ClipRect(myWindow^.portRect);   isDbl := (hf = lastFldHnd) and            (theEvent.when - lastFldTime < clickDelay);   lastFldHnd := hf;   lastFldTime := theEvent.when;   if hf <> curFld then begin      DropFocus;      curFld := hf;      InitFState(curFld, dlgFSH);      DrawField(curFld, dlgFSH, false, true);      isDbl := false   end   else      HilightSel(curFld, dlgFSH, false);   if isDbl then      SelectWord(pt, curFld, dlgFSH, nbr, blinkT)   else      Select(pt, curFld, dlgFSH, nbr, blinkT);   while StillDown do begin      GetMouse(mpt);      if not PtInRect(mpt, nbr) then         GrowSel(mpt, curFld, dlgFSH, nbr);      LetOthersRun   end;   StartBlink(blinkT);   if blinkT > 0 then nextBlink := Time + blinkTend;procedure NextField;begin   if not isCustom then begin      SetFocusAll(fldName);      exit(NextField)   end;   if curFld = fldW then SetFocusAll(fldH)   else if curFld = fldH then SetFocusAll(fldM)   else SetFocusAll(fldW)end;function ValidateCustom (var w, h, m : integer) : boolean;   function CheckOne (hf : hndField; labelPh : integer;                      lo, hi : integer; var v : integer) : boolean;   var txt, ls, los, his : Str255;       ok : boolean;   begin      FieldText(hf, txt);      ok := TextToInt(txt, v);      if ok then ok := (v >= lo) and (v <= hi);      if not ok then begin         GetPhrase(labelPh, ls);         IntToText(lo, los);         IntToText(hi, his);         ParamAlert(txt, ls, los);         ArgAlert(4, his);         StopAlert(allAlerts, alrtBadValue);         SetPort(myWindow);         ClipRect(myWindow^.portRect);         curFld := hf;         InitFState(curFld, dlgFSH);         ErrorSelect(curFld, dlgFSH, 0, hf^^.curLen, blinkT);         if blinkT > 0 then nextBlink := Time + blinkT      end;      CheckOne := ok   end;begin   ValidateCustom := false;   DropFocus;   if not CheckOne(fldW, phWidth, minCusDim, maxCusCols, w) then      exit(ValidateCustom);   if not CheckOne(fldH, phHeight, minCusDim, maxCusRows, h) then      exit(ValidateCustom);   if not CheckOne(fldM, phMines, minCusMines, w * h - 9, m) then      exit(ValidateCustom);   ValidateCustom := trueend;procedure DlgProcessEvent;var theMenu, theItem : integer;    pt        : Point;    hf        : hndField;    ferr      : integer;    ch        : char;    dummyOpen : boolean;begin   case theEvent.what of      buttonDown :         if theEvent.who = menuFolder then begin            MenuSelect(theEvent.where, theMenu, theItem);            if (theMenu = mEdit) and (theItem <> 0) then               DoDlgEditCommand(theItem)            else if (theMenu = mScrap) and (theItem <> 0) then               ScrapMenuCommand(theItem);            HiLiteMenu(0)         end         else if theEvent.who = myWindow then begin            pt := theEvent.where;            if PtInRect(pt, okR) then begin               if TrackDlgButton(okR) then begin                  dlgAccepted := true;                  dlgDone := true               end            end            else if PtInRect(pt, cancelR) then begin               if TrackDlgButton(cancelR) then begin                  dlgAccepted := false;                  dlgDone := true               end            end            else begin               hf := FieldFromPoint(pt);               if hf <> nil then DlgMouseInField(hf, pt)            end         end;      buttonUp :         ;      keyDown :         if theEvent.who = myWindow then begin            if theEvent.appleKey then begin               MenuKey(theEvent.ascii, theMenu, theItem);               if theMenu = mEdit then begin                  HiLiteMenu(theMenu);                  DoDlgEditCommand(theItem);                  HiLiteMenu(0)               end            end            else begin               ch := theEvent.ascii;               if ch = chr(13) then begin                  dlgAccepted := true;                  dlgDone := true               end               else if ch = chr(27) then begin                  dlgAccepted := false;                  dlgDone := true               end               else if ch = chr(9) then                  NextField               else if curFld <> nil then begin                  SetPort(myWindow);                  ClipRect(myWindow^.portRect);                  if ch = chr(8) then begin                     CaretOn(curFld, dlgFSH);                     Backspace(curFld, dlgFSH, ferr);                     if ferr <> 0 then BeepAlert(1);                     DrawField(curFld, dlgFSH, true, true)                  end                  else if (isCustom and (ch >= '0') and (ch <= '9')) or                          ((not isCustom) and (ch >= ' ')) then begin                     CaretOn(curFld, dlgFSH);                     InsCh(ch, curFld, dlgFSH, ferr);                     if ferr <> 0 then BeepAlert(1);                     DrawField(curFld, dlgFSH, true, true)                  end                  else                     BeepAlert(1)               end            end         end;      folderActivate :         begin            TakeControl(theEvent, false, false);            if theEvent.who = scrapFolder then               PutUpScrapMenus            else begin               PutUpDlgMenus;               if curFld <> nil then begin                  SetPort(myWindow);                  ClipRect(myWindow^.portRect);                  ChangeSelHilight(curFld, dlgFSH, false)               end            end         end;      folderDeactivate :         begin            if curFld <> nil then begin               SetPort(myWindow);               ClipRect(myWindow^.portRect);               ChangeSelHilight(curFld, dlgFSH, true)            end;            GiveControl(theEvent)         end;      folderUpdate :         if theEvent.who = scrapFolder then            DrawWindowScrap(true)         else begin            BeginUpdate(theEvent.who);            if theEvent.who = myWindow then begin               DrawContents(theEvent.who);               DrawPanel            end;            EndUpdate(theEvent.who)         end;      filerEvent :         begin            HandleFilerOp(dummyOpen);            if toldToDie or (myWindow = nil) then begin               dlgAccepted := false;               dlgDone := true            end         end;      otherwise   endend;function RunDialog : boolean;begin   dlgDone := false;   dlgAccepted := false;   lastFldHnd := nil;   lastFldTime := 0;   while not dlgDone do begin      if not ImActive then begin         GetEvent(theEvent);         DlgProcessEvent      end      else if EventAvail then begin         GetEvent(theEvent);         DlgProcessEvent      end      else begin         if (curFld <> nil) and (blinkT > 0) then            if Time >= nextBlink then begin               SetPort(myWindow);               ClipRect(myWindow^.portRect);               BlinkCaret(curFld, dlgFSH, blinkT);               if blinkT > 0 then nextBlink := Time + blinkT            end;         LetOthersRun      end   end;   RunDialog := dlgAcceptedend;procedure InitDialogs ;var trpl    : Tstr;    s       : Str255;    zeroR   : Rect;begin   err := 0;   GetPhrase(phTriples, s);   trpl := copy(s, 1, 50);   FEInit(theHeap, @trpl, err);   if err <> 0 then exit(InitDialogs);   SetRect(zeroR, 0, 0, 0, 0);   fldW    := CreateField(zeroR, numFldLen,  0, right, 4, 1, 0, fntLabel, false);   fldH    := CreateField(zeroR, numFldLen,  0, right, 4, 1, 0, fntLabel, false);   fldM    := CreateField(zeroR, numFldLen,  0, right, 4, 1, 0, fntLabel, false);   fldName := CreateField(zeroR, nameFldLen, 0, left,  4, 1, 0, fntLabel, false);   if (fldW = nil) or (fldH = nil) or (fldM = nil) or (fldName = nil) then begin      err := FEallocationFailed;      exit(InitDialogs)   end;   dlgFSH := pointer(ord(NewHandle(sizeof(fieldState))));   if dlgFSH = nil then begin      err := FEallocationFailed;      exit(InitDialogs)   end;   curFld := nil;   blinkT := -1end;function DoCustomDialog ;var s        : Str255;    finished : boolean;    accepted : boolean;begin   isCustom := true;   LayoutPanel;   IntToText(myDoc^.customW, s);   StuffField(fldW, s);   IntToText(myDoc^.customH, s);   StuffField(fldH, s);   IntToText(myDoc^.customM, s);   StuffField(fldM, s);   SetDlgMenuState(true);   DrawPanel;   SetFocusAll(fldW);   finished := false;   accepted := false;   repeat      if RunDialog then begin         if ValidateCustom(w, h, m) then begin            myDoc^.customW := w;            myDoc^.customH := h;            myDoc^.customM := m;            accepted := true;            finished := true         end      end      else         finished := true   until finished;   SetDlgMenuState(false);   DropFocus;   if (myWindow <> nil) and (not toldToDie) then      DrawContents(myWindow);   DoCustomDialog := acceptedend;function DoNameDialog ;var s : Str255;begin   isCustom := false;   LayoutPanel;   s := '';   StuffField(fldName, s);   SetDlgMenuState(true);   DrawPanel;   SetFocusAll(fldName);   if RunDialog then begin      DropFocus;      FieldText(fldName, winnerName);      DoNameDialog := true   end   else begin      DropFocus;      winnerName := '';      DoNameDialog := false   end;   SetDlgMenuState(false);   if (myWindow <> nil) and (not toldToDie) then      DrawContents(myWindow)end;end.
