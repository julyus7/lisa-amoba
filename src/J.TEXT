unit MSGame;INTERFACEuses (*$U UnitStd.obj *)        UnitStd,     (*$U UnitHz.obj *)         UnitHz,     (*$U Storage.obj *)        Storage,     (*$U QuickDraw.obj *)      QuickDraw,     (*$U Fontmgr.obj *)        FontMgr,     (*$U SysCall.obj *)        SysCall,     (*$U PMDecl.obj *)         PMDecl,     (*$U PrStdInfo.obj *)      PrStdInfo,     (*$U PrPublic.obj *)       PrPublic,     (*$U wm.events.obj *)      Events,     (*$U wm.folders.obj *)     Folders,     (*$U wm.menus.obj *)       Menus,     (*$U AlertMgr.obj *)       AlertMgr,     (*$U WMLstd.obj *)         WMLstd,     (*$U WMlCrs.obj *)         WMLCrs,     (*$U FilerComm.obj *)      FilerComm,     (*$U G.OBJ *) MSGlobals;procedure StartNewGame (newLevel : MSLevel; w, h, m : integer);procedure PlaceMines (avoidC, avoidR : integer);procedure RevealCell (c, r : integer);procedure CycleMark (c, r : integer);function AdjFlags (c, r : integer) : integer;procedure ChordCell (c, r : integer);IMPLEMENTATIONvar   workC : array [1..540] of integer;   workR : array [1..540] of integer;   workTop : integer;function RandomBelow (n : integer) : integer;var r : integer;begin   r := Random;   if r < 0 then r := -(r + 1);   RandomBelow := r mod nend;procedure StartNewGame ;var c, r : integer;begin   with myDoc^ do begin      level      := newLevel;      cols       := w;      rows       := h;      mineCount  := m;      state      := msFresh;      elapsed    := 0;      flagsPlaced:= 0;      cellsToGo  := w * h - m;      fatalCol   := 0;      fatalRow   := 0;      for r := 1 to maxRows do         for c := 1 to maxCols do begin            cells[r, c].isMine     := false;            cells[r, c].isRevealed := false;            cells[r, c].playerMark := pmNone;            cells[r, c].adjMines   := 0         end   end;   ClearDirtyend;procedure PlaceMines ;var placed  : integer;    c, r    : integer;    dc, dr  : integer;    n       : integer;    inHalo  : boolean;begin   randSeed := Time;   with myDoc^ do begin      placed := 0;      while placed < mineCount do begin         c := 1 + RandomBelow(cols);         r := 1 + RandomBelow(rows);         inHalo := (abs(c - avoidC) <= 1) and (abs(r - avoidR) <= 1);         if (not cells[r, c].isMine) and (not inHalo) then begin            cells[r, c].isMine := true;            placed := placed + 1         end      end;      for r := 1 to rows do         for c := 1 to cols do begin            n := 0;            for dr := -1 to 1 do               for dc := -1 to 1 do                  if (dc <> 0) or (dr <> 0) then                     if (c + dc >= 1) and (c + dc <= cols) and                        (r + dr >= 1) and (r + dr <= rows) then                        if cells[r + dr, c + dc].isMine then n := n + 1;            cells[r, c].adjMines := n         end;      state := msPlaying   endend;procedure RevealCell ;var dc, dr : integer;    cc, rr : integer;begin   with myDoc^ do begin      if (state = msWon) or (state = msLost) then exit(RevealCell);      if cells[r, c].playerMark = pmFlag then exit(RevealCell);      if cells[r, c].isRevealed then exit(RevealCell);      if cells[r, c].isMine then begin         cells[r, c].isRevealed := true;         fatalCol := c;         fatalRow := r;         state := msLost;         MarkDirty(c, r);         for rr := 1 to rows do            for cc := 1 to cols do               if cells[rr, cc].isMine then                  MarkDirty(cc, rr);         exit(RevealCell)      end;      cells[r, c].isRevealed := true;      cells[r, c].playerMark := pmNone;      cellsToGo := cellsToGo - 1;      MarkDirty(c, r);      workTop := 0;      if cells[r, c].adjMines = 0 then begin         workTop := 1;  workC[1] := c;  workR[1] := r      end;      while workTop > 0 do begin         cc := workC[workTop];         rr := workR[workTop];         workTop := workTop - 1;         for dr := -1 to 1 do            for dc := -1 to 1 do               if (dc <> 0) or (dr <> 0) then                  if (cc + dc >= 1) and (cc + dc <= cols) and                     (rr + dr >= 1) and (rr + dr <= rows) then                     if (not cells[rr + dr, cc + dc].isRevealed) and                        (cells[rr + dr, cc + dc].playerMark <> pmFlag) then begin                        cells[rr + dr, cc + dc].isRevealed := true;                        cells[rr + dr, cc + dc].playerMark := pmNone;                        cellsToGo := cellsToGo - 1;                        MarkDirty(cc + dc, rr + dr);                        if cells[rr + dr, cc + dc].adjMines = 0 then begin                           workTop := workTop + 1;                           workC[workTop] := cc + dc;                           workR[workTop] := rr + dr                        end                     end      end;      if cellsToGo = 0 then begin         state := msWon;         for rr := 1 to rows do            for cc := 1 to cols do               if cells[rr, cc].isMine then                  if cells[rr, cc].playerMark <> pmFlag then begin                     cells[rr, cc].playerMark := pmFlag;                     MarkDirty(cc, rr)                  end;         flagsPlaced := mineCount      end   endend;procedure CycleMark ;begin   with myDoc^ do begin      if (state = msWon) or (state = msLost) then exit(CycleMark);      if cells[r, c].isRevealed then exit(CycleMark);      MarkDirty(c, r);      case cells[r, c].playerMark of         pmNone :            begin               cells[r, c].playerMark := pmFlag;               flagsPlaced := flagsPlaced + 1            end;         pmFlag :            begin               cells[r, c].playerMark := pmQuery;               flagsPlaced := flagsPlaced - 1            end;         pmQuery :            cells[r, c].playerMark := pmNone      end   endend;function AdjFlags ;var dc, dr, n : integer;begin   n := 0;   with myDoc^ do      for dr := -1 to 1 do         for dc := -1 to 1 do            if (dc <> 0) or (dr <> 0) then               if (c + dc >= 1) and (c + dc <= cols) and                  (r + dr >= 1) and (r + dr <= rows) then                  if cells[r + dr, c + dc].playerMark = pmFlag then n := n + 1;   AdjFlags := nend;procedure ChordCell ;var dc, dr : integer;begin   with myDoc^ do begin      if state <> msPlaying then exit(ChordCell);      if not cells[r, c].isRevealed then exit(ChordCell);      if cells[r, c].adjMines = 0 then exit(ChordCell);      if AdjFlags(c, r) <> cells[r, c].adjMines then exit(ChordCell);      for dr := -1 to 1 do         for dc := -1 to 1 do            if (dc <> 0) or (dr <> 0) then               if (c + dc >= 1) and (c + dc <= cols) and                  (r + dr >= 1) and (r + dr <= rows) then                  RevealCell(c + dc, r + dr)   endend;end.
